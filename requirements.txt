# ===========================================================================
# telco-delivery-graphrag — Phase 1 RAG baseline dependencies.
# Reuses the parent ChatBOT project's proven RAG stack (LangChain + Chroma +
# FastEmbed + Anthropic), trimmed to what the baseline needs. PDF/PPTX/Streamlit
# are intentionally deferred (synthetic DMC docs are markdown/text). The Phase 2
# graph layer uses NetworkX (in-process, file-persisted — no server).
# ===========================================================================

# --- LLM provider (optional at runtime; config.json can select "none") ---
# anthropic 1.11.0's messages.create() does NOT accept `temperature`.
anthropic==1.11.0

# --- LangChain + vector store (core RAG) ---
langchain
langchain-core
langchain-community
langchain-chroma
langchain-text-splitters
chromadb

# --- Local embeddings via ONNX (no torch/transformers): BAAI/bge-small-en-v1.5
fastembed

# --- Optional local SLM provider (Qwen2.5-1.5B GGUF via llama.cpp).
# Only needed if config.json sets "llm_provider": "qwen". The .gguf is NOT shipped.
llama-cpp-python

# --- Graph store (Phase 2). In-process graph via NetworkX — no server, no
# Docker, persists to a single JSON file. Right-sized for this project's
# ~100-node knowledge graph. The backend is abstracted in src/graph.py so a
# server-backed store (e.g. Neo4j) could be swapped in later without touching
# extraction or retrieval. The Neo4j driver below is kept DORMANT (commented)
# for that optional future and is NOT required to run the project.
networkx
# neo4j   # optional future server-backed backend (not used; see src/graph.py)

# --- Utilities ---
python-dotenv
requests

# --- GUI (optional front-end). Browser-based Q&A + Impact Explorer over the
# existing UI-free src.rag / src.graph modules. Run: streamlit run app.py ---
streamlit
# Interactive knowledge-graph visualization in the Impact Explorer tab.
pyvis

# --- Dev / test ---
pytest
